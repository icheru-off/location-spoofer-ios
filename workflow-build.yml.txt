name: Build

on:
  push:
    branches: [main, master]
  workflow_dispatch:

permissions:
  contents: write

jobs:
  iphone-app:
    name: iPhone app (.ipa)
    runs-on: macos-15
    steps:
      - uses: actions/checkout@v4
      - name: Update Rust
        run: rustup update stable && rustup default stable
      - name: Build
        run: bash ios/build.sh
      - uses: actions/upload-artifact@v4
        with:
          name: LocationSpoofer-ipa
          path: build/LocationSpoofer.ipa
          if-no-files-found: error

  pairing-tool:
    name: Windows pairing tool (.exe)
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@v4
      - name: Update Rust
        run: rustup update stable && rustup default stable
      - name: Build
        run: cargo build --release -p pairtool
      - uses: actions/upload-artifact@v4
        with:
          name: LocationSpooferPair-exe
          path: target/release/LocationSpooferPair.exe
          if-no-files-found: error

  release:
    name: Publish downloads
    needs: [iphone-app, pairing-tool]
    runs-on: ubuntu-latest
    steps:
      - uses: actions/download-artifact@v4
        with:
          path: out
          merge-multiple: true
      - name: Create release
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          gh release create "build-${{ github.run_number }}" \
            out/LocationSpoofer.ipa out/LocationSpooferPair.exe \
            --repo "${{ github.repository }}" \
            --target "${{ github.sha }}" \
            --title "Build ${{ github.run_number }}" \
            --notes "LocationSpoofer.ipa : a installer sur l'iPhone avec Sideloadly. LocationSpooferPair.exe : a lancer une fois sur le PC, iPhone branche."
